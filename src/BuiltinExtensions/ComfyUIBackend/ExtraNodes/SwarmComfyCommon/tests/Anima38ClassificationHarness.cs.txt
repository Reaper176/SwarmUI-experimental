using Newtonsoft.Json.Linq;
using SwarmUI.Text2Image;
using System;
using System.Reflection;

internal static class Anima38ClassificationHarness
{
    /// <summary>Creates the minimal tensor descriptor needed by model classification predicates.</summary>
    private static JObject TensorDescriptor()
    {
        return new JObject
        {
            ["shape"] = new JArray(1)
        };
    }

    /// <summary>Creates a representative Cosmos Predict2 Text2Image 14B safetensors header.</summary>
    private static JObject CosmosHeader(bool declareArchitecture)
    {
        const string cosmosId = "nvidia-cosmos-predict2-t2i-14b";
        JObject header = new()
        {
            ["net.blocks.0.adaln_modulation_cross_attn.1.weight"] = TensorDescriptor(),
            ["net.pos_embedder.dim_temporal_range"] = TensorDescriptor(),
            ["net.x_embedder.proj.1.weight"] = TensorDescriptor(),
            ["net.blocks.35.adaln_modulation_mlp.2.weight"] = TensorDescriptor()
        };
        if (declareArchitecture)
        {
            header["__metadata__"] = new JObject
            {
                ["modelspec.architecture"] = cosmosId
            };
        }
        return header;
    }

    /// <summary>Creates an Anima 3.8B header that also carries representative Cosmos tensor names.</summary>
    private static JObject Anima38Header(bool declareArchitecture)
    {
        JObject header = CosmosHeader(declareArchitecture);
        header["net.t_embedder.1.linear_2.weight"] = TensorDescriptor();
        header["net.llm_adapter.blocks.0.self_attn.v_proj.weight"] = TensorDescriptor();
        header["net.blocks.27.adaln_modulation_cross_attn.2.weight"] = TensorDescriptor();
        header["net.blocks.51.adaln_modulation_cross_attn.2.weight"] = TensorDescriptor();
        return header;
    }

    /// <summary>Creates an older generic Anima header without the Anima 3.8B depth marker.</summary>
    private static JObject GenericAnimaHeader()
    {
        return new JObject
        {
            ["net.t_embedder.1.linear_2.weight"] = TensorDescriptor(),
            ["net.llm_adapter.blocks.0.self_attn.v_proj.weight"] = TensorDescriptor(),
            ["net.blocks.27.adaln_modulation_cross_attn.2.weight"] = TensorDescriptor()
        };
    }

    /// <summary>Creates an Anima LoRA header using one maintained tensor-key convention.</summary>
    private static JObject AnimaLoraHeader(string prefix, string suffix, bool includeMlp = true)
    {
        string key(string semanticName)
        {
            if (prefix == "lora_unet_")
            {
                return $"{prefix}{semanticName.Replace('.', '_')}{suffix}";
            }
            return $"{prefix}{semanticName}{suffix}";
        }
        JObject header = new()
        {
            [key("blocks.27.self_attn.v_proj")] = TensorDescriptor(),
            [key("blocks.27.cross_attn.output_proj")] = TensorDescriptor()
        };
        if (includeMlp)
        {
            header[key("blocks.27.mlp.layer2")] = TensorDescriptor();
        }
        return header;
    }

    /// <summary>Identifies the production model class for a synthetic safetensors model and header.</summary>
    private static string Identify(JObject header, string modelType = "Stable-Diffusion")
    {
        T2IModel model = new(null, null, "classification.safetensors", "classification");
        return T2IModelClassSorter.IdentifyClassFor(model, header, modelType)?.ID;
    }

    /// <summary>Throws when production classification differs from the expected model class.</summary>
    private static void AssertClass(string scenario, JObject header, string expected)
    {
        string actual = Identify(header);
        if (actual != expected)
        {
            throw new InvalidOperationException($"{scenario}: expected '{expected}', but production resolved '{actual ?? "null"}'.");
        }
    }

    /// <summary>Asserts the class produced for a representative LoRA header.</summary>
    private static void AssertLoraClass(string scenario, JObject header, string expected)
    {
        string actual = Identify(header, "LoRA");
        if (actual != expected)
        {
            throw new InvalidOperationException($"{scenario}: expected '{expected}', but production resolved '{actual ?? "null"}'.");
        }
    }

    /// <summary>Returns whether one compiled method directly calls another.</summary>
    private static bool CallsMethod(MethodInfo caller, MethodInfo callee)
    {
        byte[] instructions = caller.GetMethodBody()?.GetILAsByteArray()
            ?? throw new InvalidOperationException($"{caller.Name} has no method body.");
        for (int i = 0; i <= instructions.Length - 5; i++)
        {
            if (instructions[i] == 0x28 && BitConverter.ToInt32(instructions, i + 1) == callee.MetadataToken)
            {
                return true;
            }
        }
        return false;
    }

    /// <summary>Validates targeted revision invalidation for cached Cosmos classifications.</summary>
    private static void VerifyCacheRevision()
    {
        FieldInfo revisionField = typeof(T2IModelHandler).GetField("ModelClassCacheRevision", BindingFlags.NonPublic | BindingFlags.Static)
            ?? throw new InvalidOperationException("T2IModelHandler.ModelClassCacheRevision was not found.");
        MethodInfo staleMethod = typeof(T2IModelHandler).GetMethod("IsModelClassCacheStale", BindingFlags.NonPublic | BindingFlags.Static)
            ?? throw new InvalidOperationException("T2IModelHandler.IsModelClassCacheStale was not found.");
        MethodInfo advanceMethod = typeof(T2IModelHandler).GetMethod("ShouldAdvanceModelClassCacheRevisionWithoutClassification", BindingFlags.NonPublic | BindingFlags.Static)
            ?? throw new InvalidOperationException("T2IModelHandler.ShouldAdvanceModelClassCacheRevisionWithoutClassification was not found.");
        MethodInfo replaceMethod = typeof(T2IModelHandler).GetMethod("ShouldReplaceStaleModelClassCache", BindingFlags.NonPublic | BindingFlags.Static, null, [typeof(bool), typeof(bool), typeof(T2IModelClass)], null)
            ?? throw new InvalidOperationException("T2IModelHandler.ShouldReplaceStaleModelClassCache with model-header provenance was not found.");
        MethodInfo selectMethod = typeof(T2IModelHandler).GetMethod("SelectModelClassForCache", BindingFlags.NonPublic | BindingFlags.Static, null, [typeof(bool), typeof(T2IModelClass), typeof(T2IModelClass)], null)
            ?? throw new InvalidOperationException("T2IModelHandler.SelectModelClassForCache was not found.");
        MethodInfo readableMethod = typeof(T2IModelHandler).GetMethod("CanReadModelHeaderForClassification", BindingFlags.NonPublic | BindingFlags.Static)
            ?? throw new InvalidOperationException("T2IModelHandler.CanReadModelHeaderForClassification was not found.");
        PropertyInfo metadataRevision = typeof(T2IModelHandler.ModelMetadataStore).GetProperty("ModelClassRevision")
            ?? throw new InvalidOperationException("ModelMetadataStore.ModelClassRevision was not found.");
        int currentRevision = (int)revisionField.GetRawConstantValue();
        T2IModelHandler.ModelMetadataStore staleCosmos = new()
        {
            ModelClassType = "nvidia-cosmos-predict2-t2i-14b"
        };
        T2IModelHandler.ModelMetadataStore freshCosmos = new()
        {
            ModelClassType = "nvidia-cosmos-predict2-t2i-14b"
        };
        T2IModelHandler.ModelMetadataStore staleAnima = new()
        {
            ModelClassType = "anima-3_8b"
        };
        metadataRevision.SetValue(staleCosmos, currentRevision - 1);
        metadataRevision.SetValue(freshCosmos, currentRevision);
        metadataRevision.SetValue(staleAnima, currentRevision - 1);
        bool staleCosmosResult = (bool)staleMethod.Invoke(null, [staleCosmos]);
        bool freshCosmosResult = (bool)staleMethod.Invoke(null, [freshCosmos]);
        bool staleAnimaResult = (bool)staleMethod.Invoke(null, [staleAnima]);
        if (!staleCosmosResult || freshCosmosResult || staleAnimaResult)
        {
            throw new InvalidOperationException($"Unexpected cache revision policy: stale Cosmos={staleCosmosResult}, fresh Cosmos={freshCosmosResult}, stale Anima={staleAnimaResult}.");
        }
        bool advanceStaleCosmos = (bool)advanceMethod.Invoke(null, [staleCosmos]);
        bool advanceFreshCosmos = (bool)advanceMethod.Invoke(null, [freshCosmos]);
        bool advanceStaleAnima = (bool)advanceMethod.Invoke(null, [staleAnima]);
        if (advanceStaleCosmos || !advanceFreshCosmos || !advanceStaleAnima)
        {
            throw new InvalidOperationException($"Unexpected unclassified revision policy: stale Cosmos={advanceStaleCosmos}, fresh Cosmos={advanceFreshCosmos}, stale Anima={advanceStaleAnima}.");
        }
        T2IModelClass cosmosClass = T2IModelClassSorter.ModelClasses["nvidia-cosmos-predict2-t2i-14b"];
        T2IModelClass animaClass = T2IModelClassSorter.ModelClasses["anima-3_8b"];
        T2IModelClass staleSidecarOnly = (T2IModelClass)selectMethod.Invoke(null, [true, null, cosmosClass]);
        T2IModelClass staleRawAnima = (T2IModelClass)selectMethod.Invoke(null, [true, animaClass, cosmosClass]);
        T2IModelClass normalCombined = (T2IModelClass)selectMethod.Invoke(null, [false, animaClass, cosmosClass]);
        if (staleSidecarOnly is not null || staleRawAnima?.ID != "anima-3_8b" || normalCombined?.ID != "nvidia-cosmos-predict2-t2i-14b")
        {
            throw new InvalidOperationException($"Unexpected cache class selection: stale sidecar-only={staleSidecarOnly?.ID ?? "null"}, stale raw Anima={staleRawAnima?.ID ?? "null"}, normal combined={normalCombined?.ID ?? "null"}.");
        }
        bool replaceWithoutModelHeader = (bool)replaceMethod.Invoke(null, [true, false, cosmosClass]);
        bool replaceAfterSuccess = (bool)replaceMethod.Invoke(null, [true, true, cosmosClass]);
        bool replaceAfterNullClass = (bool)replaceMethod.Invoke(null, [true, true, null]);
        bool replaceNormalNull = (bool)replaceMethod.Invoke(null, [false, false, null]);
        if (replaceWithoutModelHeader || !replaceAfterSuccess || replaceAfterNullClass || !replaceNormalNull)
        {
            throw new InvalidOperationException($"Unexpected stale replacement policy: missing model header={replaceWithoutModelHeader}, successful refresh={replaceAfterSuccess}, null classifier={replaceAfterNullClass}, normal null classification={replaceNormalNull}.");
        }
        bool readsSafetensors = (bool)readableMethod.Invoke(null, ["model.safetensors"]);
        bool readsSft = (bool)readableMethod.Invoke(null, ["model.sft"]);
        bool readsGguf = (bool)readableMethod.Invoke(null, ["model.gguf"]);
        bool readsEngine = (bool)readableMethod.Invoke(null, ["model.engine"]);
        if (!readsSafetensors || !readsSft || !readsGguf || readsEngine)
        {
            throw new InvalidOperationException($"Unexpected model-header format policy: safetensors={readsSafetensors}, sft={readsSft}, gguf={readsGguf}, engine={readsEngine}.");
        }
        MethodInfo resetMetadata = typeof(T2IModelHandler).GetMethod(nameof(T2IModelHandler.ResetMetadataFrom))
            ?? throw new InvalidOperationException("T2IModelHandler.ResetMetadataFrom was not found.");
        MethodInfo loadMetadata = typeof(T2IModelHandler).GetMethod(nameof(T2IModelHandler.LoadMetadata))
            ?? throw new InvalidOperationException("T2IModelHandler.LoadMetadata was not found.");
        if (!CallsMethod(resetMetadata, advanceMethod))
        {
            throw new InvalidOperationException("ResetMetadataFrom does not preserve classifier revision provenance.");
        }
        if (!CallsMethod(loadMetadata, staleMethod) || !CallsMethod(loadMetadata, selectMethod) || !CallsMethod(loadMetadata, replaceMethod) || !CallsMethod(loadMetadata, readableMethod))
        {
            throw new InvalidOperationException("LoadMetadata does not wire stale-cache detection, raw-header selection, and guarded replacement together.");
        }
    }

    /// <summary>Validates Anima 3.8B and genuine Cosmos classification with and without declared metadata.</summary>
    private static int Main()
    {
        const string cosmosId = "nvidia-cosmos-predict2-t2i-14b";
        if (T2IModelClassSorter.ModelClasses.Count == 0)
        {
            T2IModelClassSorter.Init();
        }
        AssertClass("Anima 3.8B with declared Cosmos metadata", Anima38Header(true), "anima-3_8b");
        AssertClass("Anima 3.8B without metadata", Anima38Header(false), "anima-3_8b");
        AssertClass("Genuine Cosmos with declared Cosmos metadata", CosmosHeader(true), cosmosId);
        AssertClass("Genuine Cosmos without metadata", CosmosHeader(false), cosmosId);
        AssertClass("Older generic Anima", GenericAnimaHeader(), "anima");
        JObject differentlyDeclaredAnima = Anima38Header(false);
        differentlyDeclaredAnima["__metadata__"] = new JObject
        {
            ["modelspec.architecture"] = "stable-diffusion-v3-medium"
        };
        AssertClass("Anima tensor signature with a different known architecture", differentlyDeclaredAnima, "stable-diffusion-v3-medium");
        AssertLoraClass(
            "Anima Diffusers LoRA",
            AnimaLoraHeader("diffusion_model.", ".lora_A.weight"),
            "anima/lora");
        AssertLoraClass(
            "Anima nested Diffusers LoRA",
            AnimaLoraHeader("model.diffusion_model.", ".lora_A.default.weight"),
            "anima/lora");
        AssertLoraClass(
            "Anima Kohya LoRA",
            AnimaLoraHeader("lora_unet_", ".lora_up.weight"),
            "anima/lora");
        AssertLoraClass(
            "Incomplete Anima-like LoRA",
            AnimaLoraHeader("diffusion_model.", ".lora_A.weight", false),
            null);
        VerifyCacheRevision();
        return 0;
    }
}
